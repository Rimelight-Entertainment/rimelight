### Loading Auto

Use the `loadingAuto` prop to show the loading icon automatically while the click promise is pending.

<ComponentShowcase>
  <div class="flex flex-wrap items-center gap-md">
    <RLAButton
      id="loading-auto-demo"
      label="Submit"
      loadingAuto
    />
  </div>

  <div slot="code-astro">
    ```astro
    ---
    import RLAButton from "@rimelight/ui/components/astro/RLAButton.astro";
    ---

    <RLAButton label="Submit" loadingAuto onclick="return new Promise(resolve => setTimeout(resolve, 2000))" />
    ```

  </div>

  <div slot="code-vue">
    ```vue
    <script setup>
    import RLVButton from "@rimelight/ui/components/vue/RLVButton.vue";

    async function onClick() {
      return new Promise(resolve => setTimeout(resolve, 2000));
    }
    </script>

    <template>
      <RLVButton label="Submit" loading-auto @click="onClick" />
    </template>
    ```

  </div>
</ComponentShowcase>
