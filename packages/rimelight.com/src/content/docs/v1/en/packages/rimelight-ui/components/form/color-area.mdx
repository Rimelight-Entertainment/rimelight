The Color Area component allows users to select color values by clicking or dragging within a 2D saturation/lightness gradient box.

## Usage

How to use the Color Area component in its simplest form.

## Examples

Here is a live interactive example combining a **Color Area** with a synchronized **Color Field** showing the selected color value.

<div class="p-8 border border-border rounded-2xl bg-background/50 flex flex-col items-center justify-center gap-6 max-w-sm mx-auto shadow-lg backdrop-blur-md">
  <RLAColorArea id="demo-area" colorSpace="hsl" xChannel="saturation" yChannel="lightness" defaultValue="#3b82f6" class="w-56 h-56 rounded-xl overflow-hidden shadow-inner" />

  <div class="w-full flex flex-col gap-2">
    <label class="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Selected Color</label>
    <RLAColorField data-color-link="demo-area" id="demo-area-field" defaultValue="#3b82f6" class="w-full font-mono uppercase font-bold text-sm tracking-wide" />
  </div>
</div>

<script is:inline>{`
  const initArea = () => {
    const area = document.getElementById("demo-area");
    const field = document.getElementById("demo-area-field");
    if (area && field) {
      field.addEventListener("change", (e) => {
        area.setAttribute("data-value", e.detail);
      });
      area.addEventListener("change", (e) => {
        field.setAttribute("data-value", e.detail);
      });
    }
  };

  document.addEventListener("DOMContentLoaded", initArea);
  document.addEventListener("astro:page-load", initArea);
`}</script>
