export const demoHeaders = ["Name", "Email", "Role"]
export const demoRows = [
  ["Jane Doe", "jane@example.com", "Admin"],
  ["John Smith", "john@example.com", "Developer"],
  ["Alice Johnson", "alice@example.com", "Designer"]
]

The Table component organizes and displays structured datasets, offering built-in styling for borders, striping, and interactive hover states.

## Usage

Define a list of `headers` and `rows` using the props.

<ComponentShowcase>
  <RLATable headers={demoHeaders} rows={demoRows} />

  <div slot="code-astro">
    ```astro
    ---
    import RLATable from "@rimelight/ui/components/table/RLATable.astro";

    const headers = ["Name", "Email", "Role"];
    const rows = [
      ["Jane Doe", "jane@example.com", "Admin"],
      ["John Smith", "john@example.com", "Developer"],
      ["Alice Johnson", "alice@example.com", "Designer"]
    ];
    ---

    <RLATable headers={headers} rows={rows} />
    ```

  </div>
</ComponentShowcase>
